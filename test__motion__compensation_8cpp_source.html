20TEST(MotionCompensation, ExcludesIncompatibleCloudsWithoutChangingBatchIndices) {
21 using CloudPtr = std::shared_ptr<const TestCloud>;
22 const std::vector<int> reference_fields{1, 2, 3};
23 const CloudPtr compatible = std::make_shared<TestCloud>(TestCloud{16, reference_fields});
24 const CloudPtr incompatible_point_step = std::make_shared<TestCloud>(TestCloud{20, reference_fields});
25 const CloudPtr incompatible_fields = std::make_shared<TestCloud>(TestCloud{16, std::vector<int>{1, 2}});
26 const std::vector<CloudPtr> msgs{compatible, incompatible_point_step, nullptr, incompatible_fields};
27
28 const auto motion_msgs = selectMotionCompatibleClouds(msgs, 16, reference_fields);
29
30 ASSERT_EQ(motion_msgs.size(), msgs.size());
31 EXPECT_EQ(motion_msgs[0], compatible);
32 EXPECT_EQ(motion_msgs[1], nullptr);
33 EXPECT_EQ(motion_msgs[2], nullptr);
34 EXPECT_EQ(motion_msgs[3], nullptr);
35}
