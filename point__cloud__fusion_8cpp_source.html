81inline float loadObjectVelocity(const uint8_t* data, int datatype, bool bigendian) {
82 const uint16_t endian_check = 1;
83 uint8_t first_byte{};
84 std::memcpy(&first_byte, &endian_check, sizeof(first_byte));
85 if (bigendian != (first_byte == 0)) {
86 float value = std::numeric_limits<float>::quiet_NaN();
87 point_cloud_fusion::readScalarField(data, static_cast<uint8_t>(datatype), bigendian, value);
88 return value;
89 }
90 if (datatype == sensor_msgs::msg::PointField::FLOAT32) return loadFloat(data, 0);
91 double value{};
92 std::memcpy(&value, data, sizeof(value));
93 return std::abs(value) <= std::numeric_limits<float>::max() ? static_cast<float>(value)
94 : std::numeric_limits<float>::quiet_NaN();
95}

122template <typename Lookup>
123auto lookupWithWallTimeout(double timeout_sec, const Lookup& lookup) {
124 const auto deadline = std::chrono::steady_clock::now() + std::chrono::duration<double>(timeout_sec);
125 for (;;) {
126 try {
127 return lookup();
128 } catch (const tf2::TransformException&) {
129 if (!rclcpp::ok() || std::chrono::steady_clock::now() >= deadline) throw;
130 std::this_thread::sleep_for(std::chrono::milliseconds(1));
131 }
132 }

147inline void copyConvertedFields(const uint8_t* source,
148 uint8_t* destination,
149 const point_cloud_fusion::detail::InputLayout& input,
150 bool source_bigendian,
151 bool destination_bigendian) {
152 for (const auto& conversion : input.conversions) {
153 float value = 0.0F;
154 point_cloud_fusion::readScalarField(byteOffset(source, conversion.source_offset), conversion.source_datatype,
155 source_bigendian, value);
156 // Integer fields cannot represent unknown floating measurements.
157 if (!std::isfinite(value) && conversion.destination_datatype != sensor_msgs::msg::PointField::FLOAT32 &&
158 conversion.destination_datatype != sensor_msgs::msg::PointField::FLOAT64) {
159 value = 0.0F;
160 }
161 point_cloud_fusion::writeScalarField(byteOffset(destination, conversion.destination_offset), conversion.destination_datatype,
162 destination_bigendian, value);
163 }

582 if constexpr (std::is_integral_v<T>) {

584 range.set__from_value(static_cast<T>(from_value.value())).set__to_value(static_cast<T>(to_value.value()));
585 if (step_value.has_value()) range.set__step(static_cast<T>(step_value.value()));
586 param_desc.integer_range = {range};
587 } else if constexpr (std::is_floating_point_v<T>) {
588 rcl_interfaces::msg::FloatingPointRange range;
589 range.set__from_value(static_cast<T>(from_value.value())).set__to_value(static_cast<T>(to_value.value()));
590 if (step_value.has_value()) range.set__step(static_cast<T>(step_value.value()));
591 param_desc.floating_point_range = {range};
592 } else {
593 RCLCPP_WARN(this->get_logger(),
594 "Parameter type of parameter '%s' does not support "
595 "specifying a range",
596 name.c_str());
597 }
598 }
599
600 this->declare_parameter(name, type, param_desc);
601
602 try {
603 param = this->get_parameter(name).get_value<T>();
604 std::stringstream ss;
605 ss << "Loaded parameter '" << name << "': ";
606 if constexpr (is_vector_v<T>) {
607 ss << "[";
608 for (std::size_t i = 0; i < param.size(); ++i) ss << (i > 0 ? ", " : "") << param[i];
609 ss << "]";
610 } else {
611 ss << param;
612 }
613 RCLCPP_INFO_STREAM(this->get_logger(), ss.str());
614 } catch (rclcpp::exceptions::ParameterUninitializedException&) {
615 if (is_required) {
616 RCLCPP_FATAL_STREAM(this->get_logger(), "Missing required parameter '" << name << "', exiting");
617 exit(EXIT_FAILURE);
618 } else {
619 std::stringstream ss;
620 ss << "Missing parameter '" << name << "', using default value: ";
621 if constexpr (is_vector_v<T>) {
622 ss << "[";
623 for (std::size_t i = 0; i < param.size(); ++i) ss << (i > 0 ? ", " : "") << param[i];
624 ss << "]";
625 } else {
626 ss << param;
627 }
628 RCLCPP_WARN_STREAM(this->get_logger(), ss.str());
629 this->set_parameters({rclcpp::Parameter(name, rclcpp::ParameterValue(param))});
630 }
631 }
632
633 if (add_to_auto_reconfigurable_params) {
634 std::function<void(const rclcpp::Parameter&)> setter = [&param](const rclcpp::Parameter& p) { param = p.get_value<T>(); };
635 auto_reconfigurable_params_.push_back(std::make_tuple(name, setter));
636 }

639rcl_interfaces::msg::SetParametersResult PointCloudFusion::parametersCallback(const std::vector<rclcpp::Parameter>& parameters) {
640 std::unique_lock<std::shared_mutex> config_lock(config_mutex_);
641
642 // Pre-validate interdependent and runtime-sensitive parameters before
643 // applying any changes.
644 // Build the prospective state: current values overridden by incoming changes.
645 std::optional<detail::MissingFieldFillValues> prospective_fills;
646 bool prospective_object_motion = object_motion_compensation_enable_;
647 bool prospective_ego_motion = motion_compensation_enable_;
648 bool any_range_param = false;
649 bool prospective_use_cuda = use_cuda_;
650 int64_t prospective_fixed_points_per_input_cloud = fixed_points_per_input_cloud_;
651 double prospective_x_min = range_limits_x_min_;
652 double prospective_x_max = range_limits_x_max_;
653 double prospective_y_min = range_limits_y_min_;
654 double prospective_y_max = range_limits_y_max_;
655 double prospective_z_min = range_limits_z_min_;
656 double prospective_z_max = range_limits_z_max_;
657
658 for (const auto& param : parameters) {
659 const auto& name = param.get_name();
660 if (name == "missing_field_fill_values") {
661 try {
662 prospective_fills = detail::parseMissingFieldFillValues(param.as_string_array());
663 } catch (const std::exception& error) {
664 rcl_interfaces::msg::SetParametersResult result;
665 result.successful = false;
666 result.reason = error.what();
667 return result;
668 }
669 } else if (name == "object_motion_compensation.enable") {
670 prospective_object_motion = param.as_bool();
671 } else if (name == "motion_compensation.enable") {
672 prospective_ego_motion = param.as_bool();
673 } else if (name == "fixed_points_per_input_cloud") {
674 prospective_fixed_points_per_input_cloud = param.as_int();
675 } else if (name == "use_cuda") {
676 prospective_use_cuda = param.as_bool();
677 } else if (name == "range_limits.x_min") {
678 prospective_x_min = param.as_double();
679 any_range_param = true;
680 } else if (name == "range_limits.x_max") {
681 prospective_x_max = param.as_double();
682 any_range_param = true;
683 } else if (name == "range_limits.y_min") {
684 prospective_y_min = param.as_double();
685 any_range_param = true;
686 } else if (name == "range_limits.y_max") {
687 prospective_y_max = param.as_double();
688 any_range_param = true;
689 } else if (name == "range_limits.z_min") {
690 prospective_z_min = param.as_double();
691 any_range_param = true;
692 } else if (name == "range_limits.z_max") {
693 prospective_z_max = param.as_double();
694 any_range_param = true;
695 }
696 }
697
698 if (prospective_object_motion && !prospective_ego_motion) {
699 rcl_interfaces::msg::SetParametersResult result;
700 result.successful = false;
701 result.reason = "object_motion_compensation.enable requires motion_compensation.enable";
702 return result;
703 }
704
705 if (prospective_fixed_points_per_input_cloud < kMinFixedPointsPerInputCloud ||
706 prospective_fixed_points_per_input_cloud > kMaxFixedPointsPerInputCloud) {
707 rcl_interfaces::msg::SetParametersResult result;
708 result.successful = false;
709 result.reason = "fixed_points_per_input_cloud must be in [" + std::to_string(kMinFixedPointsPerInputCloud) + ", " +
710 std::to_string(kMaxFixedPointsPerInputCloud) + "]";
711 RCLCPP_ERROR(this->get_logger(), "Rejecting parameter update: %s", result.reason.c_str());
712 return result;
713 }
714
715 if (prospective_use_cuda) {
716#ifdef ENABLE_CUDA
717 if (!cuda_context_) {
718 rcl_interfaces::msg::SetParametersResult result;
719 result.successful = false;
720 result.reason =
721 "CUDA backend is unavailable because CUDA context "
722 "initialization failed";
723 RCLCPP_ERROR(this->get_logger(), "Rejecting parameter update: %s", result.reason.c_str());
724 return result;
725 }
726#else
727 rcl_interfaces::msg::SetParametersResult result;
728 result.successful = false;
729 result.reason = "CUDA backend is unavailable because CUDA support was not compiled";
730 RCLCPP_ERROR(this->get_logger(), "Rejecting parameter update: %s", result.reason.c_str());
731 return result;
732#endif
733 }
734
735 if (any_range_param) {
736 rcl_interfaces::msg::SetParametersResult result;
737 std::string reason;
738 if (prospective_x_min >= prospective_x_max) {
739 reason += "range_limits.x_min (" + std::to_string(prospective_x_min) + ") must be less than range_limits.x_max (" +
740 std::to_string(prospective_x_max) + "). ";
741 }
742 if (prospective_y_min >= prospective_y_max) {
743 reason += "range_limits.y_min (" + std::to_string(prospective_y_min) + ") must be less than range_limits.y_max (" +
744 std::to_string(prospective_y_max) + "). ";
745 }
746 if (prospective_z_min >= prospective_z_max) {
747 reason += "range_limits.z_min (" + std::to_string(prospective_z_min) + ") must be less than range_limits.z_max (" +
748 std::to_string(prospective_z_max) + "). ";
749 }
750 if (!reason.empty()) {
751 result.successful = false;
752 result.reason = reason;
753 RCLCPP_ERROR(this->get_logger(), "Rejecting range_limits parameter update: %s", reason.c_str());
754 return result;
755 }
756 }
757
758 // All validations passed — apply changes.
759 const bool previous_use_cuda = use_cuda_;
760 for (const auto& param : parameters) {
761 for (auto& auto_reconfigurable_param : auto_reconfigurable_params_) {
762 if (param.get_name() == std::get<0>(auto_reconfigurable_param)) {
763 std::get<1>(auto_reconfigurable_param)(param);
764 RCLCPP_INFO(this->get_logger(), "Reconfigured parameter '%s' to: %s", param.get_name().c_str(),
765 param.value_to_string().c_str());
766 break;
767 }
768 }
769 }
770 if (prospective_fills) missing_field_fills_ = std::move(*prospective_fills);
771 if (use_cuda_ != previous_use_cuda) {
772 RCLCPP_INFO(this->get_logger(), "Fusion backend switched to %s", use_cuda_ ? "CUDA" : "CPU");
773 }
774
775 rcl_interfaces::msg::SetParametersResult result;
776 result.successful = true;
777
778 return result;
779}

897void PointCloudFusion::setup() {
898 // callback for dynamic parameter configuration
899 parameters_callback_ =
900 this->add_on_set_parameters_callback(std::bind(&PointCloudFusion::parametersCallback, this, std::placeholders::_1));
901
902 // create transform buffer and listener
903 tf_buffer_ = std::make_shared<tf2_ros::Buffer>(this->get_clock());
904 tf_listener_ = std::make_shared<tf2_ros::TransformListener>(*tf_buffer_);
905
906 // validate inputs
907 if (!input_transport_hints_.empty() && input_transport_hints_.size() != input_topics_.size()) {
908 RCLCPP_WARN(this->get_logger(),
909 "'input_transport_hints' length (%zu) does not match "
910 "'input_topics' (%zu). Missing hints default to "
911 "'%s'",
912 input_transport_hints_.size(), input_topics_.size(), kDefaultTransportHint);
913 }
914
915 // create subscribers
916 cloud_subscribers_.clear();
917 cloud_subscriber_callback_groups_.clear();
918 cloud_subscribers_.reserve(input_topics_.size());
919 cloud_subscriber_callback_groups_.reserve(input_topics_.size());
920 for (size_t i = 0; i < input_topics_.size(); ++i) {
921 const std::string configured = input_topics_[i];
922 const std::string resolved = this->get_node_topics_interface()->resolve_topic_name(configured);
923 const std::string hint = (i < input_transport_hints_.size() && !input_transport_hints_[i].empty())
924 ? input_transport_hints_[i]
925 : std::string(kDefaultTransportHint);
926
927 auto callback_group = this->create_callback_group(rclcpp::CallbackGroupType::MutuallyExclusive);
928 rclcpp::SubscriptionOptions subscription_options;
929 subscription_options.callback_group = callback_group;
930
931 auto subscriber = std::make_shared<point_cloud_transport::SubscriberFilter>();
932 // Point clouds are sensor data and are commonly published best-effort to
933 // avoid back-pressure on high-bandwidth streams. A best-effort subscriber
934 // remains compatible with reliable publishers, so it also supports mixed
935 // sensor setups (for example, best-effort Aeva and reliable Ouster).
936 subscriber->subscribe(this->shared_from_this(), resolved, hint, rmw_qos_profile_sensor_data, subscription_options);
937 RCLCPP_INFO(this->get_logger(), "Subscribed to '%s' (hint=%s)", subscriber->getTopic().c_str(), hint.c_str());
938 cloud_subscriber_callback_groups_.push_back(std::move(callback_group));
939 cloud_subscribers_.push_back(std::move(subscriber));
940 }
941
942 RCLCPP_INFO(this->get_logger(), "Configured %zu input subscriber callback groups for %zu input topics",
943 cloud_subscriber_callback_groups_.size(), input_topics_.size());
944
945 synchronizer_.reset();
946
947 // configure synchronization or direct passthrough
948 if (cloud_subscribers_.size() == 1) {
949 cloud_subscribers_.front()->registerCallback([this](const PointCloudMsg::ConstSharedPtr msg) {
950 std::vector<PointCloudMsg::ConstSharedPtr> batch;
951 batch.reserve(1);
952 batch.emplace_back(msg);
953 this->handleSynchronizedPointClouds(batch);
954 });
955 RCLCPP_INFO(this->get_logger(), "Configured single-input mode for topic '%s'",
956 cloud_subscribers_.front()->getTopic().c_str());
957 } else if (cloud_subscribers_.size() <= 9) {
958 switch (cloud_subscribers_.size()) {
959 case 2:
960 setupSynchronizer<2>();
961 break;
962 case 3:
963 setupSynchronizer<3>();
964 break;
965 case 4:
966 setupSynchronizer<4>();
967 break;
968 case 5:
969 setupSynchronizer<5>();
970 break;
971 case 6:
972 setupSynchronizer<6>();
973 break;
974 case 7:
975 setupSynchronizer<7>();
976 break;
977 case 8:
978 setupSynchronizer<8>();
979 break;
980 case 9:
981 setupSynchronizer<9>();
982 break;
983 default:
984 RCLCPP_FATAL(this->get_logger(), "Unsupported number of input topics: %zu", cloud_subscribers_.size());
985 exit(EXIT_FAILURE);
986 }
987 }
988
989 // create publisher
990 point_cloud_transport::PointCloudTransport pct(this->shared_from_this());
991 std::string point_cloud_topic_name = this->get_node_topics_interface()->resolve_topic_name("~/point_cloud");
992 cloud_publisher_ = std::make_shared<point_cloud_transport::Publisher>(
993 pct.advertise(point_cloud_topic_name, static_cast<uint32_t>(output_queue_size_)));
994 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", cloud_publisher_->getTopic().c_str());
995
996 // Annotate message links for tracing: Each publisher (for raw and compressed
997 // point clouds) depends an all input point clouds.
998 std::vector<const void*> link_subs;
999 std::vector<const void*> link_pubs;
1000 for (const auto& sub_filter : cloud_subscribers_) {
1001 auto sub_base = sub_filter->getSubscriber().getSubscription();
1002 if (sub_base) {
1003 link_subs.push_back(static_cast<const void*>(sub_base->get_subscription_handle().get()));
1004 }
1005 }
1006 if (cloud_publisher_) {
1007 std::map<std::string, rclcpp::PublisherBase::SharedPtr> pubs_base = cloud_publisher_->getPublishers();
1008 for (const auto& [transport, pub_base] : pubs_base) {
1009 link_pubs.push_back(static_cast<const void*>(pub_base->get_publisher_handle().get()));
1010 }
1011 }
1012 TRACETOOLS_TRACEPOINT(message_link_partial_sync, link_subs.data(), link_subs.size(), link_pubs.data(), link_pubs.size());
1013}

1015template <std::size_t N>
1016void PointCloudFusion::setupSynchronizer() {
1017 static_assert(N >= 2 && N <= 9, "Supported synchronizer size is between 2 and 9");
1018
1019 using Policy = detail::SyncPolicy<N>;
1020 using Sync = detail::SyncType<N>;
1021
1022 // Instantiate ApproximateTime policy tuned to the active input count.
1023 auto sync = std::make_shared<Sync>(Policy(static_cast<size_t>(sync_queue_size_)));
1024
1025 // Wire the configured subscribers into the synchronizer slots.
1026 detail::connectInputs<N>(*sync, cloud_subscribers_);
1027
1028 sync->setMaxIntervalDuration(rclcpp::Duration::from_seconds(max_time_diff_sec_));
1029 sync->setAgePenalty(age_penalty_);
1030 sync->registerCallback([this](auto&&... msgs) {
1031 std::vector<PointCloudMsg::ConstSharedPtr> batch;
1032 batch.reserve(sizeof...(msgs));
1033 // Extract only valid PointCloud2 pointers from the variadic callback.
1034 auto append = [&batch](auto&& msg) {
1035 using ArgT = std::decay_t<decltype(msg)>;
1036 if constexpr (std::is_same_v<ArgT, PointCloudMsg::ConstSharedPtr>) {
1037 if (msg) {
1038 batch.emplace_back(std::forward<decltype(msg)>(msg));
1039 }
1040 }
1041 };
1042 (append(std::forward<decltype(msgs)>(msgs)), ...);
1043 if (!batch.empty()) {
1044 this->handleSynchronizedPointClouds(batch);
1045 } else {
1046 RCLCPP_WARN(this->get_logger(),
1047 "ApproximateTime synchronizer yielded no "
1048 "valid point clouds; skipping fusion.");
1049 }
1050 });
1051
1052 synchronizer_ = sync;
1053
1054 RCLCPP_INFO(this->get_logger(),
1055 "Configured approximate time synchronizer for %zu inputs "
1056 "(queue=%zu, max_dt=%.3f s, age_penalty=%.6f)",
1057 static_cast<size_t>(N), static_cast<size_t>(sync_queue_size_), max_time_diff_sec_, age_penalty_);
1058}

1060void PointCloudFusion::handleSynchronizedPointClouds(const std::vector<sensor_msgs::msg::PointCloud2::ConstSharedPtr>& msgs) {
1061 if (msgs.empty()) {
1062 return;
1063 }
1064
1065 if (feature_calibrator_) feature_calibrator_->enqueue(msgs);
1066
1067 // Protect runtime-configurable parameter reads against concurrent parameter
1068 // updates.
1069 std::shared_lock<std::shared_mutex> config_lock(config_mutex_);
1070
1071 const auto callback_start = std::chrono::steady_clock::now();
1072
1073 FusionTiming timing;
1074 if (!collectTimingInfo(msgs, timing)) {
1075 return;
1076 }
1077
1078#ifdef ENABLE_CUDA
1079 std::size_t cuda_valid_count = 0;
1080 auto processing_start = std::chrono::steady_clock::time_point{};
1081 auto processing_end = std::chrono::steady_clock::time_point{};
1082 PointCloudMsg::UniquePtr cuda_result;
1083 bool used_cuda = false;
1084 {
1085 // Guard shared CUDA pipeline state against concurrent synchronized
1086 // callbacks.
1087 std::lock_guard<std::mutex> cuda_lock(cuda_context_mutex_);
1088 // Run either CPU or CUDA implementation based on parameter.
1089 if (cuda_context_ && use_cuda_) {
1090 used_cuda = true;
1091 processing_start = std::chrono::steady_clock::now();
1092 cuda_result = fusePointCloudBatchCUDA(msgs, timing, cuda_valid_count);
1093 processing_end = std::chrono::steady_clock::now();
1094 }
1095 }
1096
1097 if (used_cuda) {
1098 if (cuda_result) {
1099 publishFusedCloud(std::move(cuda_result), timing, msgs.size(), cuda_valid_count, callback_start, processing_start,
1100 processing_end, "cuda_fusion_complete");
1101 } else {
1102 RCLCPP_WARN(this->get_logger(), "CUDA processing failed");
1103 }
1104 return;
1105 }
1106#endif
1107
1108 // CPU-only path
1109 std::size_t valid_count = 0;
1110 const auto cpu_processing_start = std::chrono::steady_clock::now();
1111 auto fused_point_cloud = fusePointCloudBatch(msgs, timing, valid_count);
1112
1113 if (!fused_point_cloud) {
1114 RCLCPP_WARN(this->get_logger(), "All points are invalid, skipping fusion");
1115 return;
1116 }
1117
1118 const auto cpu_processing_end = std::chrono::steady_clock::now();
1119 publishFusedCloud(std::move(fused_point_cloud), timing, msgs.size(), valid_count, callback_start, cpu_processing_start,
1120 cpu_processing_end, "cpu_fusion_complete");
1121}

1123bool PointCloudFusion::collectTimingInfo(const std::vector<PointCloudMsg::ConstSharedPtr>& msgs, FusionTiming& timing) const {
1124 if (msgs.empty()) {
1125 return false;
1126 }
1127
1128 const auto input0_stamp = rclcpp::Time(msgs.front()->header.stamp);
1129 bool first_stamp = true;
1130 double max_dt_from_input0_sec = 0.0;
1131 rclcpp::Time earliest_stamp;
1132 rclcpp::Time latest_stamp;
1133
1134 // Walk every cloud once to gather min/max stamps and the largest skew from
1135 // input0.
1136 for (const auto& pc_msg : msgs) {
1137 if (!pc_msg) {
1138 RCLCPP_WARN(this->get_logger(),
1139 "Received null point cloud pointer in "
1140 "synchronized batch, skipping fusion");
1141 return false;
1142 }
1143
1144 const rclcpp::Time current_stamp(pc_msg->header.stamp);
1145 if (first_stamp) {
1146 earliest_stamp = current_stamp;
1147 latest_stamp = current_stamp;
1148 first_stamp = false;
1149 } else {
1150 if (current_stamp < earliest_stamp) earliest_stamp = current_stamp;
1151 if (current_stamp > latest_stamp) latest_stamp = current_stamp;
1152 }
1153
1154 const double dt_sec = std::fabs((current_stamp - input0_stamp).seconds());
1155 if (dt_sec > max_dt_from_input0_sec) {
1156 max_dt_from_input0_sec = dt_sec;
1157 }
1158 }
1159
1160 timing.earliest_stamp = earliest_stamp;
1161 timing.latest_stamp = latest_stamp;
1162 timing.input0_stamp = input0_stamp;
1163 timing.max_dt_from_input0_sec = max_dt_from_input0_sec;
1164 return true;
1165}

1167rclcpp::Time PointCloudFusion::outputStamp(const FusionTiming& timing) const {
1168 switch (output_stamp_mode_) {
1169 case OutputStampMode::Earliest:
1170 return timing.earliest_stamp;
1171 case OutputStampMode::Mean: {
1172 const auto delta = timing.latest_stamp - timing.earliest_stamp;
1173 return timing.earliest_stamp + rclcpp::Duration::from_nanoseconds(delta.nanoseconds() / 2);
1174 }
1175 case OutputStampMode::Input0:
1176 return timing.input0_stamp;
1177 case OutputStampMode::Latest:
1178 default:
1179 return timing.latest_stamp;
1180 }
1181}

1183bool PointCloudFusion::prepareMotionTransform(const PointCloudMsg& msg,
1184 const rclcpp::Time& reference_stamp,
1185 int time_field_offset,
1186 double timeout_sec,
1187 MotionTransform& transform) const {
1188 transform = MotionTransform{};
1189
1190 const std::size_t total_points = static_cast<std::size_t>(msg.width) * static_cast<std::size_t>(msg.height);
1191 if (time_field_offset >= 0 && total_points > 0) {
1192 const auto offset = static_cast<std::size_t>(time_field_offset);
1193 if (offset + sizeof(uint32_t) <= msg.point_step && msg.data.size() >= total_points * msg.point_step) {
1194 const auto* data = msg.data.data();
1195 uint32_t max_time_offset = 0;
1196 for (std::size_t idx = 0; idx < total_points; ++idx) {
1197 max_time_offset = std::max(max_time_offset, loadUint32(byteOffset(data, idx * msg.point_step), offset));
1198 }
1199 transform.max_time_offset = max_time_offset;
1200 }
1201 }
1202
1203 // A cloud already expressed in the fixed output frame needs no TF lookup,
1204 // even when that frame has no entries in the buffer.
1205 if (msg.header.frame_id == target_frame_ && target_frame_ == motion_compensation_fixed_frame_) return true;
1206
1207 const rclcpp::Time scan_start(msg.header.stamp);
1208 const auto scan_duration =
1209 rclcpp::Duration::from_seconds(static_cast<double>(transform.max_time_offset) * motion_compensation_time_scale_sec_);
1210 const rclcpp::Time scan_end = scan_start + scan_duration;
1211 auto lookup_at = [&](const rclcpp::Time& source_stamp) {
1212 return lookupWithWallTimeout(timeout_sec, [&] {
1213 return tf_buffer_->lookupTransform(target_frame_, reference_stamp, msg.header.frame_id, source_stamp,
1214 motion_compensation_fixed_frame_, rclcpp::Duration::from_seconds(0.0));
1215 });
1216 };
1217
1218 geometry_msgs::msg::TransformStamped start_stamped;
1219 geometry_msgs::msg::TransformStamped end_stamped;
1220 try {
1221 start_stamped = lookup_at(scan_start);
1222 end_stamped = transform.max_time_offset > 0 ? lookup_at(scan_end) : start_stamped;
1223 } catch (const tf2::TransformException& ex) {
1224 RCLCPP_WARN_THROTTLE(this->get_logger(), *this->get_clock(), 5000,
1225 "Motion compensation unavailable for '%s' through "
1226 "fixed frame '%s': %s; "
1227 "falling back to rigid transforms for this batch",
1228 msg.header.frame_id.c_str(), motion_compensation_fixed_frame_.c_str(), ex.what());
1229 return false;
1230 }
1231
1232 auto copy_transform = [](const geometry_msgs::msg::Transform& source, std::array<float, 3>& translation,
1233 std::array<float, 4>& quaternion) {
1234 translation[0] = static_cast<float>(source.translation.x);
1235 translation[1] = static_cast<float>(source.translation.y);
1236 translation[2] = static_cast<float>(source.translation.z);
1237 quaternion[0] = static_cast<float>(source.rotation.x);
1238 quaternion[1] = static_cast<float>(source.rotation.y);
1239 quaternion[2] = static_cast<float>(source.rotation.z);
1240 quaternion[3] = static_cast<float>(source.rotation.w);
1241 };
1242 copy_transform(start_stamped.transform, transform.start_translation, transform.start_quaternion);
1243 copy_transform(end_stamped.transform, transform.end_translation, transform.end_quaternion);
1244
1245 const float quaternion_dot =
1246 transform.start_quaternion[0] * transform.end_quaternion[0] + transform.start_quaternion[1] * transform.end_quaternion[1] +
1247 transform.start_quaternion[2] * transform.end_quaternion[2] + transform.start_quaternion[3] * transform.end_quaternion[3];
1248 if (quaternion_dot < 0.0F) {
1249 for (float& component : transform.end_quaternion) {
1250 component = -component;
1251 }
1252 }
1253 return true;
1254}

1256bool PointCloudFusion::prepareBatchMotionTransforms(const std::vector<PointCloudMsg::ConstSharedPtr>& msgs,
1257 const rclcpp::Time& reference_stamp,
1258 const std::vector<int>& time_field_offsets,
1259 std::vector<MotionTransform>& transforms) const {
1260 transforms.clear();
1261 transforms.resize(msgs.size());
1262 if (!motion_compensation_enable_ || motion_compensation_fixed_frame_.empty()) {
1263 return false;
1264 }
1265
1266 if (time_field_offsets.size() != msgs.size()) return false;
1267
1268 const bool recovery_probe = !motion_tf_available_.load(std::memory_order_relaxed);
1269 const double timeout_sec = recovery_probe ? 0.0 : motion_compensation_tf_timeout_sec_;
1270 bool prepared_any = false;
1271 for (std::size_t i = 0; i < msgs.size(); ++i) {
1272 if (!msgs[i] || time_field_offsets[i] == -2) {
1273 continue;
1274 }
1275 prepared_any = true;
1276 // Deskewed XYZ needs only the header pose, but radial object motion still
1277 // needs the acquisition origin expressed in that header frame.
1278 const bool deskewed = i < motion_compensation_deskewed_inputs_.size() && motion_compensation_deskewed_inputs_[i];
1279 const bool needs_ray_origin =
1280 deskewed && object_motion_compensation_enable_ && prepareObjectMotion(*msgs[i], i, reference_stamp).velocity_offset >= 0;
1281 const int deskew_time_offset = deskewed && !needs_ray_origin ? -1 : time_field_offsets[i];
1282 if (!prepareMotionTransform(*msgs[i], reference_stamp, deskew_time_offset, timeout_sec, transforms[i])) {
1283 motion_tf_available_.store(false, std::memory_order_relaxed);
1284 return false;
1285 }
1286 if (deskewed) retainDeskewedPose(transforms[i]);
1287 }
1288
1289 if (prepared_any && !motion_tf_available_.exchange(true, std::memory_order_relaxed)) {
1290 RCLCPP_INFO(this->get_logger(),
1291 "Motion-compensation TF is available "
1292 "again; resuming compensated fusion");
1293 }
1294 return prepared_any;
1295}

1316ObjectMotion PointCloudFusion::prepareObjectMotion(const PointCloudMsg& cloud,
1317 std::size_t input_index,
1318 const rclcpp::Time& reference_stamp) const {
1319 ObjectMotion motion;
1320 if (!object_motion_compensation_enable_) return motion;
1321 const auto& name =
1322 object_motion_velocity_fields_.empty() ? object_motion_velocity_field_ : object_motion_velocity_fields_[input_index];
1323 if (name.empty()) return motion;
1324 const auto* velocity = detail::findField(cloud, name);
1325 if (velocity == nullptr) return motion; // Missing velocity is expected for sensors such as Ouster.
1326 if (velocity->count != 1 || !detail::validField(*velocity, cloud.point_step) ||
1327 (velocity->datatype != sensor_msgs::msg::PointField::FLOAT32 &&
1328 velocity->datatype != sensor_msgs::msg::PointField::FLOAT64)) {
1329 RCLCPP_WARN_THROTTLE(this->get_logger(), *this->get_clock(), 5000,
1330 "Object motion: '%s' must be a scalar FLOAT32/FLOAT64 measurement; skipping correction", name.c_str());
1331 return motion;
1332 }
1333 if (object_motion_invalid_flags_mask_ != 0) {
1334 const auto* flags = detail::findField(cloud, object_motion_invalid_flags_field_);
1335 if (flags == nullptr || flags->count != 1 || flags->datatype != sensor_msgs::msg::PointField::UINT32 ||
1336 !detail::validField(*flags, cloud.point_step)) {
1337 RCLCPP_WARN_THROTTLE(this->get_logger(), *this->get_clock(), 5000,
1338 "Object motion: configured invalid_flags_field is unavailable; skipping correction");
1339 return motion;
1340 }
1341 motion.invalid_flags_offset = static_cast<int>(flags->offset);
1342 motion.invalid_flags_mask = static_cast<uint32_t>(object_motion_invalid_flags_mask_);
1343 }
1344 motion.velocity_offset = static_cast<int>(velocity->offset);
1345 motion.velocity_datatype = velocity->datatype;
1346 // Subtract integer ROS timestamps before converting the short interval to float.
1347 motion.reference_offset_sec = static_cast<float>((reference_stamp - rclcpp::Time(cloud.header.stamp)).seconds());
1348 motion.time_scale_sec = static_cast<float>(motion_compensation_time_scale_sec_);
1349 motion.doppler_sign = static_cast<float>(object_motion_doppler_sign_);
1350 motion.max_time_offset_sec = static_cast<float>(object_motion_max_time_offset_sec_);
1351 motion.max_speed_mps = static_cast<float>(object_motion_max_speed_mps_);
1352 return motion;
1353}

1385
1386PointCloudFusion::PointCloudMsg::UniquePtr PointCloudFusion::fusePointCloudBatch(
1387 const std::vector<PointCloudMsg::ConstSharedPtr>& msgs, const FusionTiming& timing, std::size_t& valid_point_count) const {
1388 valid_point_count = 0;
1389 if (msgs.empty()) return nullptr;
1390
1391 const auto calibration_field = calibrationOutputField(msgs);
1392 const detail::BatchLayout layout =
1393 detail::buildBatchLayout(msgs, output_fields_, motion_compensation_time_field_, missing_field_fills_,
1394 calibration_field ? &*calibration_field : nullptr);
1395 if (!layout.fill_error.empty()) {
1396 RCLCPP_ERROR_THROTTLE(this->get_logger(), *this->get_clock(), 5000, "%s; skipping batch", layout.fill_error.c_str());
1397 return nullptr;
1398 }
1399 if (layout.point_step == 0) {
1400 missing_xyz_count_.fetch_add(msgs.size(), std::memory_order_relaxed);
1401 RCLCPP_WARN(this->get_logger(), "No input cloud has valid FLOAT32 x/y/z fields; skipping batch");
1402 return nullptr;
1403 }
1404 warnUnusedFillOverrides(layout.unused_fill_fields);
1405
1406 if (!layout.conflicting_fields.empty()) {
1407 incompatible_field_count_.fetch_add(layout.conflicting_fields.size(), std::memory_order_relaxed);
1408 std::ostringstream names;
1409 for (std::size_t i = 0; i < layout.conflicting_fields.size(); ++i) {
1410 if (i != 0U) names << ", ";
1411 names << layout.conflicting_fields[i];
1412 }
1413 RCLCPP_WARN_THROTTLE(this->get_logger(), *this->get_clock(), 5000,
1414 "Omitting same-name fields with incompatible datatype/count: %s", names.str().c_str());
1415 }
1416
1417 std::size_t max_capacity = 0;
1418 std::vector<int> time_offsets(msgs.size(), -2);
1419 for (std::size_t i = 0; i < msgs.size(); ++i) {
1420 if (!layout.inputs[i].valid || !msgs[i]) {
1421 if (msgs[i]) {
1422 missing_xyz_count_.fetch_add(1, std::memory_order_relaxed);
1423 skipped_cloud_count_.fetch_add(1, std::memory_order_relaxed);
1424 RCLCPP_WARN_THROTTLE(this->get_logger(), *this->get_clock(), 5000,
1425 "Skipping cloud from '%s': %s (missing_xyz_total=%llu, "
1426 "skipped_total=%llu)",
1427 msgs[i]->header.frame_id.c_str(), layout.inputs[i].rejection_reason.c_str(),
1428 static_cast<unsigned long long>(missing_xyz_count_.load()),
1429 static_cast<unsigned long long>(skipped_cloud_count_.load()));
1430 }
1431 continue;
1432 }
1433 time_offsets[i] = layout.inputs[i].time_offset;
1434 std::size_t count = static_cast<std::size_t>(msgs[i]->width) * msgs[i]->height;
1435 count = std::min(count, msgs[i]->point_step == 0 ? std::size_t{0} : msgs[i]->data.size() / msgs[i]->point_step);
1436 if (fixed_points_per_input_cloud_ > 0) {
1437 count = std::min(count, static_cast<std::size_t>(fixed_points_per_input_cloud_));
1438 }
1439 max_capacity += count;
1440 if (!layout.inputs[i].zero_filled_fields.empty()) {
1441 zero_filled_field_count_.fetch_add(layout.inputs[i].zero_filled_fields.size(), std::memory_order_relaxed);
1442 RCLCPP_DEBUG(this->get_logger(), "Cloud from '%s' zero-fills %zu output fields (total=%llu)",
1443 msgs[i]->header.frame_id.c_str(), layout.inputs[i].zero_filled_fields.size(),
1444 static_cast<unsigned long long>(zero_filled_field_count_.load()));
1445 }
1446 }
1447 if (max_capacity == 0) return nullptr;
1448
1449 const auto calibration_mappings =
1450 feature_calibrator_ ? feature_calibrator_->mappings() : std::vector<std::shared_ptr<const FeatureMapping>>(msgs.size());
1451 const auto output_calibration_field = std::find_if(layout.fields.begin(), layout.fields.end(), [this](const auto& field) {
1452 return field.name == feature_calibration_field_ && field.count == 1;
1453 });
1454
1455 auto output = std::make_unique<PointCloudMsg>();
1456 output->header.frame_id = target_frame_;
1457 const rclcpp::Time chosen_stamp = outputStamp(timing);
1458 output->header.stamp = chosen_stamp;
1459 output->height = 1;
1460 output->is_bigendian = layout.is_bigendian;
1461 output->point_step = layout.point_step;
1462 output->fields = layout.fields;
1463 output->is_dense = layout.is_dense;
1464 output->data.resize(max_capacity * layout.point_step);
1465
1466 const bool check_range = range_limits_enable_;
1467 const float x_min = static_cast<float>(range_limits_x_min_);
1468 const float x_max = static_cast<float>(range_limits_x_max_);
1469 const float y_min = static_cast<float>(range_limits_y_min_);
1470 const float y_max = static_cast<float>(range_limits_y_max_);
1471 const float z_min = static_cast<float>(range_limits_z_min_);
1472 const float z_max = static_cast<float>(range_limits_z_max_);
1473
1474 std::vector<MotionTransform> motion_transforms;
1475 const bool batch_motion = prepareBatchMotionTransforms(msgs, chosen_stamp, time_offsets, motion_transforms);
1476 int output_time_offset = -1;
1477 for (const auto& field : layout.fields) {
1478 if (field.name == motion_compensation_time_field_ && field.datatype == sensor_msgs::msg::PointField::UINT32 &&
1479 field.count == 1) {
1480 output_time_offset = static_cast<int>(field.offset);
1481 break;
1482 }
1483 }
1484 uint8_t* destination = output->data.data();
1485
1486 for (std::size_t input_index = 0; input_index < msgs.size(); ++input_index) {
1487 const auto& msg = msgs[input_index];
1488 const auto& input = layout.inputs[input_index];
1489 if (!msg || !input.valid || msg->point_step == 0) continue;
1490 const auto mapping = input_index < calibration_mappings.size() ? calibration_mappings[input_index] : nullptr;
1491 const auto* input_calibration_field = mapping ? detail::findField(*msg, feature_calibration_field_) : nullptr;
1492 const bool calibrate = mapping && input_calibration_field != nullptr && input_calibration_field->count == 1 &&
1493 detail::validField(*input_calibration_field, msg->point_step) &&
1494 output_calibration_field != layout.fields.end();
1495
1496 const bool apply_transform = batch_motion || msg->header.frame_id != target_frame_;
1497 auto object_motion = batch_motion ? prepareObjectMotion(*msg, input_index, chosen_stamp) : ObjectMotion{};
1498 object_motion.ray_origin_max_time_offset = motion_transforms[input_index].ray_origin_max_time_offset;
1499 object_motion.end_ray_origin = motion_transforms[input_index].end_ray_origin;
1500 tf2::Vector3 translation(0.0, 0.0, 0.0);
1501 tf2::Matrix3x3 rotation = tf2::Matrix3x3::getIdentity();
1502 if (!batch_motion && apply_transform) {
1503 try {
1504 const auto stamped = lookupWithWallTimeout(0.1, [&] {
1505 return tf_buffer_->lookupTransform(target_frame_, msg->header.frame_id, msg->header.stamp,
1506 rclcpp::Duration::from_seconds(0.0));
1507 });
1508 tf2::Transform transform;
1509 tf2::fromMsg(stamped.transform, transform);
1510 translation = transform.getOrigin();
1511 rotation = transform.getBasis();
1512 } catch (const tf2::TransformException& exception) {
1513 skipped_cloud_count_.fetch_add(1, std::memory_order_relaxed);
1514 RCLCPP_ERROR(this->get_logger(),
1515 "Cannot transform point cloud from %s to %s: %s "
1516 "(skipped_total=%llu)",
1517 msg->header.frame_id.c_str(), target_frame_.c_str(), exception.what(),
1518 static_cast<unsigned long long>(skipped_cloud_count_.load()));
1519 continue;
1520 }
1521 }
1522
1523 const std::size_t declared_points = static_cast<std::size_t>(msg->width) * msg->height;
1524 const std::size_t total_points = std::min(declared_points, msg->data.size() / msg->point_step);
1525 const std::size_t samples = fixed_points_per_input_cloud_ > 0
1526 ? std::min(total_points, static_cast<std::size_t>(fixed_points_per_input_cloud_))
1527 : total_points;
1528 if (samples == 0) continue;
1529 const double stride = static_cast<double>(total_points) / static_cast<double>(samples);
1530
1531 for (std::size_t sample = 0; sample < samples; ++sample) {
1532 const std::size_t point_index =
1533 samples == total_points ? sample
1534 : std::min(static_cast<std::size_t>(static_cast<double>(sample) * stride), total_points - 1);
1535 const uint8_t* source = byteOffset(msg->data.data(), point_index * msg->point_step);
1536 float x = loadFloat(source, input.x_offset);
1537 float y = loadFloat(source, input.y_offset);
1538 float z = loadFloat(source, input.z_offset);
1539 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z)) continue;
1540
1541 const uint32_t point_time =
1542 batch_motion && input.time_offset >= 0 ? loadUint32(source, static_cast<std::size_t>(input.time_offset)) : 0U;
1543 if (object_motion.velocity_offset >= 0 &&
1544 (object_motion.invalid_flags_offset < 0 ||
1545 (loadUint32(source, static_cast<std::size_t>(object_motion.invalid_flags_offset)) &
1546 object_motion.invalid_flags_mask) == 0U)) {
1547 const float velocity = loadObjectVelocity(byteOffset(source, static_cast<std::size_t>(object_motion.velocity_offset)),
1548 object_motion.velocity_datatype, msg->is_bigendian);
1549 compensateObjectMotion(object_motion, velocity, point_time, x, y, z);
1550 }
1551
1552 float transformed_x = x;
1553 float transformed_y = y;
1554 float transformed_z = z;
1555 if (batch_motion) {
1556 transformPointInterpolated(motion_transforms[input_index], point_time, x, y, z, transformed_x, transformed_y,
1557 transformed_z);
1558 } else if (apply_transform) {
1559 const tf2::Vector3 transformed = rotation * tf2::Vector3(x, y, z) + translation;
1560 transformed_x = static_cast<float>(transformed.x());
1561 transformed_y = static_cast<float>(transformed.y());
1562 transformed_z = static_cast<float>(transformed.z());
1563 }
1564 if (check_range &&
1565 !pointWithinRange(transformed_x, transformed_y, transformed_z, x_min, x_max, y_min, y_max, z_min, z_max)) {
1566 continue;
1567 }
1568
1569 if (input.whole_point_copy) {
1570 std::memcpy(destination, source, layout.point_step);
1571 } else {
1572 std::memset(destination, 0, layout.point_step);
1573 detail::fillMissingFields(destination, input);
1574 for (const auto& copy : input.copies) {
1575 std::memcpy(byteOffset(destination, copy.destination_offset), byteOffset(source, copy.source_offset), copy.byte_length);
1576 }
1577 copyConvertedFields(source, destination, input, msg->is_bigendian, layout.is_bigendian);
1578 }
1579 storeFloat(destination, layout.x_offset, transformed_x);
1580 storeFloat(destination, layout.y_offset, transformed_y);
1581 storeFloat(destination, layout.z_offset, transformed_z);
1582 if (output_time_offset >= 0 && input.time_offset >= 0) {
1583 const uint32_t output_time =
1584 batch_motion ? 0U
1585 : rebasedTimeOffset(loadUint32(source, static_cast<std::size_t>(input.time_offset)), msg->header.stamp,
1586 chosen_stamp, motion_compensation_time_scale_sec_);
1587 storeUint32(destination, static_cast<std::size_t>(output_time_offset), output_time);
1588 }
1589 if (calibrate) {
1590 float value = 0.0F;
1591 if (readScalarField(byteOffset(source, input_calibration_field->offset), input_calibration_field->datatype,
1592 msg->is_bigendian, value)) {
1593 writeScalarField(byteOffset(destination, output_calibration_field->offset), output_calibration_field->datatype,
1594 output->is_bigendian, mapping->apply(value));
1595 }
1596 }
1597 destination = byteOffset(destination, layout.point_step);
1598 ++valid_point_count;
1599 }
1600 }
1601
1602 if (valid_point_count == 0) return nullptr;
1603 output->width = valid_point_count;
1604 output->row_step = output->point_step * output->width;
1605 output->data.resize(valid_point_count * output->point_step);
1606 return output;
1607}

1609void PointCloudFusion::publishFusedCloud(PointCloudMsg::UniquePtr cloud,
1610 const FusionTiming& timing,
1611 std::size_t input_count,
1612 std::size_t total_points,
1613 std::chrono::steady_clock::time_point callback_start,
1614 std::chrono::steady_clock::time_point processing_start,
1615 std::chrono::steady_clock::time_point processing_end,
1616 const char* event_name) {
1617 // Publish the fused cloud and emit a compact timing summary for
1618 // observability.
1619 cloud_publisher_->publish(std::move(cloud));
1620 const auto publish_end = std::chrono::steady_clock::now();
