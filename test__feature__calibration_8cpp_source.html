19sensor_msgs::msg::PointCloud2::SharedPtr makeFloatCloud(const std::vector<float>& values) {
20 auto cloud = std::make_shared<sensor_msgs::msg::PointCloud2>();
21 cloud->height = 1;
22 cloud->width = values.size();
23 cloud->point_step = 4 * sizeof(float);
24 cloud->row_step = cloud->width * cloud->point_step;
25 for (std::size_t index = 0; index < 4; ++index) {
26 sensor_msgs::msg::PointField field;
27 field.name = std::vector<std::string>{"x", "y", "z", "reflectivity"}[index];
28 field.offset = index * sizeof(float);
29 field.datatype = sensor_msgs::msg::PointField::FLOAT32;
30 field.count = 1;
31 cloud->fields.push_back(field);
32 }
33 cloud->data.resize(values.size() * cloud->point_step);
34 for (std::size_t point = 0; point < values.size(); ++point) {
35 const std::array<float, 4> record{1.0F, 2.0F, 3.0F, values[point]};
36 std::memcpy(cloud->data.data() + point * cloud->point_step, record.data(), cloud->point_step);
37 }
38 return cloud;
39}

108TEST(FeatureCalibration, WorkerPublishesFollowerMappingAsynchronously) {
109 DistributionFeatureCalibrator::Config config;
110 config.field = "reflectivity";
111 config.leaders = {0};
112 config.followers = {1};
113 config.input_count = 3;
114 config.samples_per_cloud = 4;
115 config.window_samples = 16;
116 config.minimum_samples = 4;
117 config.quantile_count = 4;
118 config.update_interval_sec = 0.001;
119 DistributionFeatureCalibrator calibrator(config);
120 const std::vector<sensor_msgs::msg::PointCloud2::ConstSharedPtr> clouds{makeFloatCloud({100.0F, 120.0F, 140.0F, 160.0F}),
121 makeFloatCloud({0.0F, 10.0F, 20.0F, 30.0F}),
122 makeFloatCloud({500.0F, 500.0F, 500.0F, 500.0F})};
123
124 std::shared_ptr<const FeatureMapping> mapping;
125 for (int attempt = 0; attempt < 100 && !mapping; ++attempt) {
126 calibrator.enqueue(clouds);
127 std::this_thread::sleep_for(std::chrono::milliseconds(2));
128 mapping = calibrator.mappings()[1];
129 }
130
131 ASSERT_TRUE(mapping);
132 EXPECT_FLOAT_EQ(mapping->apply(0.0F), 100.0F);
133 EXPECT_FLOAT_EQ(mapping->apply(30.0F), 160.0F);
134 EXPECT_EQ(calibrator.mappings()[0], nullptr);
135 EXPECT_EQ(calibrator.mappings()[2], nullptr);
136}
