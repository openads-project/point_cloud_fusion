31template <typename PointCloudPtr, typename Fields>
32std::vector<PointCloudPtr> selectMotionCompatibleClouds(const std::vector<PointCloudPtr>& msgs,
33 std::size_t point_step,
34 const Fields& fields) {
35 std::vector<PointCloudPtr> motion_msgs = msgs;
36 for (auto& motion_msg : motion_msgs) {
37 if (motion_msg && (motion_msg->point_step != point_step || motion_msg->fields != fields)) {
38 motion_msg.reset();

47inline void transformPointInterpolated(const MotionTransform& transform,
48 uint32_t time_offset,
49 float x,
50 float y,
51 float z,
52 float& transformed_x,
53 float& transformed_y,
54 float& transformed_z) {
55 const float alpha = transform.max_time_offset > 0
56 ? std::min(1.0F, static_cast<float>(time_offset) / static_cast<float>(transform.max_time_offset))
57 : 0.0F;
58 const float inverse_alpha = 1.0F - alpha;
59
60 float qx = inverse_alpha * transform.start_quaternion[0] + alpha * transform.end_quaternion[0];
61 float qy = inverse_alpha * transform.start_quaternion[1] + alpha * transform.end_quaternion[1];
62 float qz = inverse_alpha * transform.start_quaternion[2] + alpha * transform.end_quaternion[2];
63 float qw = inverse_alpha * transform.start_quaternion[3] + alpha * transform.end_quaternion[3];
64 const float inverse_norm = 1.0F / std::sqrt(qx * qx + qy * qy + qz * qz + qw * qw);
65 qx *= inverse_norm;
66 qy *= inverse_norm;
67 qz *= inverse_norm;
68 qw *= inverse_norm;
69
70 // Rotate with q * p * conjugate(q), avoiding temporary tf2 objects.
71 const float tx = 2.0F * (qy * z - qz * y);
72 const float ty = 2.0F * (qz * x - qx * z);
73 const float tz = 2.0F * (qx * y - qy * x);
74 transformed_x =
75 x + qw * tx + (qy * tz - qz * ty) + inverse_alpha * transform.start_translation[0] + alpha * transform.end_translation[0];
76 transformed_y =
77 y + qw * ty + (qz * tx - qx * tz) + inverse_alpha * transform.start_translation[1] + alpha * transform.end_translation[1];
78 transformed_z =
79 z + qw * tz + (qx * ty - qy * tx) + inverse_alpha * transform.start_translation[2] + alpha * transform.end_translation[2];
80}
